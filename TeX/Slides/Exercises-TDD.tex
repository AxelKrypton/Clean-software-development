\begin{frame}{Your tasks: 90 minutes}
    \begin{enumerate}
        \setcounter{enumi}{-1}
        \item You will work alone: Pick one programming language
        \item \PS{Read the tasks and choose an appealing one}\\
              \then Start basic if you are new to TDD
        \item Think about TDD rules\\
              \then Decide how purist you want to be
        \item \PP{Solve the problem using TDD}\\
              \then It will feel clumsy and slow...
        \item Check where you stand after 30-45 minutes
        \item Once you are done start over from scratch again\\
              \then \alert{Now you can focus on the discipline more}
    \end{enumerate}
