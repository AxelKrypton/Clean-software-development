\begin{frame}{Review another project}
    \begin{enumerate}
        \item Take the project of another group \Remark{from yesterday}\\
              \then you will review it as a group\\
              \then you have \textasciitilde60 minutes time
        \item Have a rough look to the whole code \Remark{Alone}\\
              \then note down your comments and questions
        \item Do the review together as a group\\
              \then a simple Markdown file is enough\\
              \then stick to the principles we discussed today\\
              \then classify your comments as major or minor
    \end{enumerate}
