            When you work mostly with senior developers
        \end{varblock}
        \begin{varblock}{example}[0.7\textwidth]{When does Git-flow work best?}

\begin{frame}{Issues}{\alt<1>{A way to organise your (team's) work}{This is how it might look like}}
    \vspace{-6mm}
    \begin{varblock}{example}[0.85\textwidth]{When opening the issue}<only@1>
        \begin{enumerate}
            \item Choose a descriptive title
            \item Spend time on description!
            \item Assign somebody whenever possible
            \item Consistently use labels for issues properties
            \item Consistently use types to describe the issue entity
            \item Assign the issue to a project if reasonable
            \item Add the issue to a milestone if it makes sense
        \end{enumerate}
    \end{varblock}

\MakeLabelledSubsectionWithCoverFrame{PrFrame}{Pull requests}
%===================================================================================================%
\begin{frame}{Pull requests}{A way to contribute to a repository}
    \vspace{-2mm}
    \begin{varblock}{example}[0.95\textwidth]{A clever merge procedure}
        \small
        A pull request is a way to prepare a merge of a Git branch into another one.
        Through a graphic interface, the changes can be reviewed and actions can be done.
    \end{varblock}
    \vspace{2mm}
    \begin{overlayarea}{\textwidth}{0.5\textheight}
        \begin{onlyenv}<1>
            \begin{center}
                \includegraphics[width=0.8\textwidth]{GitHub-PR}
            \end{center}
        \end{onlyenv}
        \begin{onlyenv}<2>
            \begin{itemize}
                \item Always assign at least one reviewer
                \item A PR can be opened as a \textbf{Draft}
                \item Use automatic linking to issues in the description
                \item \alert{Actions automatically start}
                \item The review is done in the ``Files changed'' tab
                \item \PB{The reviewer comments on code and can request changes}
            \end{itemize}
        \end{onlyenv}
    \end{overlayarea}
    \FrameRemark{Assignees are people in charge of merging the PR and may be who opened the PR, but also somebody else.}
\end{frame}

\begin{frame}[fragile]{Actions and continuous integration}{\only<1-2>{A world plenty of opportunities!}\only<3-4>{Some typical examples}\only<5>{Standard GitHub-hosted runners for \textbf{public} repositories}}
    \begin{onlyenv}<1>
        \begin{center}
            \includegraphics[width=0.95\textwidth]{Meme-actions}
        \end{center}
    \end{onlyenv}
    \begin{onlyenv}<2>
        \begin{lstlisting}[style=MyBash]
            # List of all possible events that can trigger an action
            branch_protection_rule         page_build
            check_run                      public
            check_suite                    pull_request
            create                         pull_request_comment
            delete                         pull_request_review
            deployment                     pull_request_target
            deployment_status              push
            discussion                     registry_package
            discussion_comment             release
            fork                           repository_dispatch
            gollum                         schedule
            issue_comment                  status
            issues                         watch
            label                          workflow_call
            merge_group                    workflow_dispatch
            milestone                      workflow_run
        \end{lstlisting}
        \begin{center}
            Find out more about \URL[PB]{https://docs.github.com/en/actions/writing-workflows/choosing-when-your-workflow-runs/events-that-trigger-workflows}{events that trigger a workflow}!
        \end{center}
    \end{onlyenv}
    \begin{onlyenv}<3-4>
        \definecolor{myCyan}{cmyk}{1,0,0,0}
        \begin{varblock}{alert}[0.85\textwidth]{}<only@3>
            \alert{Actions use YAML files to configure what needs to be done!}
        \end{varblock}
        \begin{enumerate}
            \item<only@3> Run tests for default PR activity\\[-1mm]
            \Remark{The workflow runs when a pull request is opened or reopened or when the head branch of the pull request is updated}
            \begin{lstlisting}[style=MyBash, xleftmargin=1mm, xrightmargin=25mm, aboveskip=\medskipamount,
                               moredelim={[is][\color{myCyan}]{|-}{-|}},
                               moredelim={[is][\color{ForestGreen}]{|=}{=|}}]
                |-on-|:
                  |-pull_request-|:
                    |-branches-|:
                      - |=main=|
                      - |=release*=|
            \end{lstlisting}
            \item<only@3> Run tests when merged code into some branches
            \begin{lstlisting}[style=MyBash, xleftmargin=1mm, xrightmargin=25mm,
                               belowskip=-3mm, aboveskip=\medskipamount,
                               moredelim={[is][\color{myCyan}]{|-}{-|}},
                               moredelim={[is][\color{ForestGreen}]{|=}{=|}}]
                |-on-|:
                  |-push-|:
                    |-branches-|: [|=main=|, |=develop=|, |=release*=|]
            \end{lstlisting}
            \item<only@4> Selecting GitHub virtual machines and run something
            \begin{lstlisting}[style=MyBash, xleftmargin=1mm, xrightmargin=25mm,
                               belowskip=-6mm, aboveskip=1.5\medskipamount,
                               moredelim={[is][\color{myCyan}]{|-}{-|}},
                               moredelim={[is][\color{ForestGreen}]{|=}{=|}}]
                |-jobs-|:
                  |-Run-on-Ubuntu-|:
                    |-name-|: "Test on latest Ubuntu"
                    |-runs-on-|: |=ubuntu-latest=|
                    |-steps-|:
                      - |-uses-|: |=actions/checkout@v4=|
                      - |-name-|: "Print a greeting"
                        |-env-|:
                          |-MY_VAR-|: "Hi there"
                        |-run-|: |
                          |=echo "${MY_VAR}!"=|
            \end{lstlisting}
            \item<only@4> Using \URL[PB]{https://docs.github.com/en/actions/hosting-your-own-runners/managing-self-hosted-runners/about-self-hosted-runners}{a self-hosted runner}
            \begin{lstlisting}[style=MyBash, xleftmargin=1mm, xrightmargin=25mm,
                               belowskip=-8mm, aboveskip=1.5\medskipamount,
                               moredelim={[is][\color{myCyan}]{|-}{-|}},
                               moredelim={[is][\color{ForestGreen}]{|=}{=|}}]
                |-runs-on-|: [|=self-hosted=|, |=linux=|, |=x64=|]
            \end{lstlisting}
            \Remark{If you specify an array of strings or variables, your workflow will execute on any runner that matches all of the specified \,\texttt{runs-on}\, values.}
        \end{enumerate}
    \end{onlyenv}
    \begin{onlyenv}<5>
        \begin{center}
            \vspace{2mm}
            \includegraphics[height=0.7\textheight]{GitHub-actions-servers-public}
        \end{center}
    \end{onlyenv}
    \PrepareURLsymbol[PB]
    \FrameRemark<4-5>{Checkout \URL*{https://docs.github.com/en/actions/writing-workflows/workflow-syntax-for-github-actions}{the official documentation} to discover how to structure your actions. Check out existing actions, too!}
\end{frame}
