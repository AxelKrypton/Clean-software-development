\begin{frame}{Solving Ising with MCMC}
    \begin{varblock*}{example}[0.9\textwidth]{Common strategy:}
        \hspace{0.05\textwidth}
        \begin{minipage}{0.8\textwidth}
            Generate $\nconf$ configurations randomly,\\
            \then then estimate expectation value of observable $X$
            $$
              \bar{X}=\frac{1}{\nconf}\sum_{i=1}^{\nconf} X_i
            $$
            Each $X_i=X(C_i)$ measured on $C_i$ from \textbf{Markov Chain}
            $$
              C_0\to C_1\to C_2\to...
            $$
            Each $C_i$ drawn from $\textrm{pr}(C)$
        \end{minipage}
    \end{varblock*}
\end{frame}

\begin{frame}{Physics of the Ising model}
Order parameter is \textbf{magnetization} $m$
$$
m\equiv\frac{1}{V}\sum_i\sigma_i~~~~~~~~
\chi\equiv\frac{V}{T}\left(\langle|m|^2\rangle-\langle|m|\rangle^2\right)
$$

\vspace{-3mm}
In $V\to\infty$ limit, $m$ gets a kink and $\chi$ diverges
at the \textbf{critical temperature} $T_c$.
\vspace{3mm}
%At finite $V$, kink softens to curve, divergence to peak.
