\begin{frame}{Extracting $T_c$}
    \begin{columns}
        \begin{column}{0.9\textwidth}
            The \textbf{Binder cumulant} is defined as
            $$
            B\equiv1-\frac{\langle m^4\rangle}{3\langle m^2\rangle^2}
            $$
            \begin{itemize}
            \item $B$ computed at different $L$ \textbf{will cross at} $T_c$
            \item $B=0$ for $T\gg T_c$
            \item $B=2/3$ for $T\ll T_c$
            \item Easier to get $T_c$ than using $\chi$
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}
