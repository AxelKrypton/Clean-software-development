\frame{\frametitle{Validation strategies}
\begin{varblock}{alert}[0.8\textwidth]{}
    Here are some ideas to check consistency of code output.\\
    \alert{I think of them as Cross Checks (CC).}\\
    Some are the \textbf{same checks you do in physics}.
\end{varblock}
\begin{varblock}{}[0.7\textwidth]{What I use most commonly:}
    \begin{enumerate}
    \item Compare against trusted result
    % can be from literature, another code base, etc
    \item Compute a result multiple ways
    \item Check for consistency between outputs
    % for example if one obs is the derivative of the other,
    % that constrains how they must be related
    \item Check known properties
    % probabilities in [0,1]; 2d ising magnetization monotonic in T
    \item Verify independence of hyperparameters
    % dependence on jackknife bin size? solve tolerance?
    \end{enumerate}
\end{varblock}
\begin{center}
    \footnotesize
    The list above is not meant to have a particular order, is not exhaustive,
    and is not disjoint!
\end{center}
}

\frame{\frametitle{CC1: Compare against trusted result}
\begin{varblock}{alert}[0.9\textwidth]{Danger with reproducing: you \textbf{assume other calculation is correct}}
\begin{itemize}
\item Has the result already been reproduced?
\item You could apply some Cross Checks to the trusted result
\item
\URL*{https://www.annualreviews.org/content/journals/10.1146/annurev.nucl.55.090704.151521}{Blinding}
can help against human bias:
\begin{enumerate}
\item Let $r\in[-R,R]$ unknown to you
\item Rescale your observable by $r$
\item Remove $r$ when you are confident in your calculation
\end{enumerate}
\end{itemize}
\end{varblock}

\begin{frame}[fragile]{CC2: Compute a result multiple ways}
\begin{varblock}{example}[\textwidth]{Ex: Pure Python vs Numpy}
$
\textrm{acint}(T)=1+2\sum_{t=1}^{T-1}\frac{\textrm{autocor}(t)}{\textrm{autocor}(0)}
$

\vspace{2mm}

\begin{lstlisting}[style=mypython]
# Want 2-d array. One axis for jackknife bin. One axis for T.
acintj = [[1.]*Nbins]
for t in range(1,Nt+1):
    tauintbins = []
    for bin in range(Nbins):
        tauintbins.append(
            acintj[t-1][bin] + 2*acorj[t][bin]/acorj[0][bin]
        )
    acintj.append(tauintbins)
\end{lstlisting}

\vspace{0.5cm}

\begin{lstlisting}[style=mypython]
# Faster numpy implementation. Compare acintj from both methods!
acintj     = np.ones((Nt+1,Nbins))
acintj[1:] = np.cumsum(2*acorj[1:]/acorj[0],axis=0) + 1
\end{lstlisting}
\end{varblock}
\end{frame}


\frame{\frametitle{CC3: Check for consistency between outputs}
Observables may be constrained by relationships between them.
Design tests that ensure observables obey these known constraints.
\begin{varblock}{example}[0.7\textwidth]{Examples}
\begin{itemize}
\item Speed, velocity, acceleration
\item $n_B$ and $p$ as function of $\mu_B$
\item Gibbs-Duhem: $Ts=\epsilon+p-\sum_i\mu_in_i$
\item $f\left(f^{-1}\left(x\right)\right)=x$
\end{itemize}
\end{varblock}
}


\begin{frame}[fragile]{CC3: Check for consistency between outputs}
\begin{varblock}{example}[\textwidth]{Ex: Round-trip test}
Sometimes you will transform your data, for example when switching
between different formats. A \textbf{round-trip} test checks that
data remain consistent and accurate through these transformations.
\vspace{3mm}
\begin{lstlisting}[style=mypython]
# Make test table
ref = latexTable()
ref.append(['1','2','3'])
ref.append(['1','2','3'])
ref.writeTable('testInterface.tex')

# Round-trip test
convertTable('testInterface.tex','testInterface.csv',
             targetDelimiter=',')
test.readTable('testInterface.csv',delimiter=',')
lpass *= test==ref
\end{lstlisting}
\end{varblock}
\end{frame}


\frame{\frametitle{CC4: Check known properties}
\centering
Depending on an observable's interpretation, you may
already \\ know some properties it must have. Test them!
\begin{varblock}{example}[0.85\textwidth]{Examples}
\begin{itemize}
\item is your covariance matrix positive semidefinite?
\item is your unitary matrix actually unitary?
\item is your RHMC reversible?
\item is your probability between 0 and 1?
\item are physical observables at physical parameters real?
\end{itemize}
\end{varblock}
}


\begin{frame}[fragile]{CC4: Check known properties}
\begin{varblock}{example}[0.85\textwidth]{Ex: A check for SU$(N)$ class methods}
\begin{lstlisting}[style=mypython]
class SUN(np.ndarray):
    ...
    def isSUN(self) -> bool:
        """ Check that I have det=1 and am unitary. """
        if not (isSpecial(self) and isUnitary(self)):
            return False
        return True

class SU3(SUN):
    ...

g = SU3()
g.setToRandom()

lpass *= g.isSUN()
\end{lstlisting}
\end{varblock}
\end{frame}


\frame{\frametitle{CC5: Verify independence of hyperparameters}
\begin{itemize}
\item Parameter: Directly \textbf{model-relevant} number
\begin{itemize}
\item Temperature, volume, particle mass, coupling constant
\item Fit parameters
\end{itemize}
\item Hyperparameter: Needed \textbf{only for computational strategy}
\begin{itemize}
\item Jackknife bins, bootstrap samples
\item Initial guess of a fitter
\item Max iterations, solve tolerance, CNN kernel size
\end{itemize}
\item Often only independent of hyperparameter \textbf{in some region}
\begin{itemize}
\item Number of jackknife bins until error bar plateaus
\item Initial guesses leading to reasonable $\chi^2/{\rm d.o.f.}$
\end{itemize}
\end{itemize}
}
